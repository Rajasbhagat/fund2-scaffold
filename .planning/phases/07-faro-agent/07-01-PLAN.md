// --- Helper: extract text from PDF ---
async function extractPDF(filename: string): Promise<string> {
  const filepath = path.join(SYLLABI_DIR, filename);
  const buffer = fs.readFileSync(filepath);
  const data = await pdfParse(buffer);
  return data.text.trim();
}

// --- Helper: write file ---
function writeContext(outputName: string, content: string): void {
  const dest = path.join(OUTPUT_DIR, outputName);
  fs.writeFileSync(dest, content, 'utf-8');
  const tokens = Math.round(content.length / 4);
  console.log(`Written: ${outputName} (~${tokens} tokens, ${content.length} chars)`);
}

// Falls back to a structured truncation if no recognizable structure is found.
function makeTier2Summary(name: string, professor: string, term: string, focus: string, rawText: string): string {
  // Extract topics: look for "objectives", "topics", "content", "sessions" sections
  const lines = rawText.split('\n').filter(l => l.trim().length > 0);
  // Grab first 60 lines of content for topic extraction (skip blank lines)
  const firstChunk = lines.slice(0, 60).join('\n');

  return `=== ${name} — ${professor} ===
Term: ${term}
Focus: ${focus}
---
${firstChunk.slice(0, 1200)}
[... condensed for token budget ...]`;
}

```typescript
export function getFAROContext(): string {
  const faroDir = path.join(process.cwd(), 'src', 'context', 'faro');

  // Priority header — tells the model to consult pre-loaded knowledge first
  const priorityHeader = `[FARO KNOWLEDGE BASE — FUND II Course Navigator]
Consult the pre-loaded syllabi below BEFORE searching the web.
Tier 1 (FUND II Core) is your primary authority for all course navigation questions.
Tier 2 (IESE Ecosystem) provides elective and ecosystem context for career guidance queries.
Always cite which session, syllabus, or document your answer draws from.`;

  // FARO system prompt (extracted from FARO_Agent_Spec_v2.txt)
  const faroSystemPrompt = fs.readFileSync(
    path.join(faroDir, 'faro-system-prompt.txt'),
    'utf-8'
  );

  // Tier 1 — FUND II Core (always inject in full)
  const fundIISyllabus = fs.readFileSync(
    path.join(faroDir, 'fund-ii-syllabus.txt'),
    'utf-8'
  );

  // Tier 1 — FUND I Syllabi (for FUND I bridge questions)
  const fundIFiles = ['fund-i-hietaniemi.txt', 'fund-i-snihur.txt', 'fund-i-nairobi.txt'];
  const fundIDocs = fundIFiles.map((f) => {
    try {
      return fs.readFileSync(path.join(faroDir, f), 'utf-8');
    } catch {
      return `[${f} not found — skip]`;
    }
  });

  // Tier 2 — Elective Syllabi (condensed summaries)
  const tier2Files = [
    'navei.txt',
    'vcic.txt',
    'search.txt',
    'bmi.txt',
    'bmc.txt',
    'corporate-entrepreneurship.txt',
    'sei.txt',
    'entrepreneurial-finance.txt',
    'creativity.txt',
    'sports.txt',
    'eic.txt',
  ];
  const tier2Docs = tier2Files.map((f) => {
    try {
      return fs.readFileSync(path.join(faroDir, f), 'utf-8');
    } catch {
      return `[${f} not found — skip]`;
    }
  });

  // Assemble: priority header + system prompt + FUND II syllabus + FUND I + Tier 2
  const separator = '\n\n---FARO KNOWLEDGE BASE DOCUMENT---\n\n';

  return [
    priorityHeader,
    faroSystemPrompt,
    `=== TIER 1: FUND II SYLLABUS (PRIMARY AUTHORITY) ===\n\n${fundIISyllabus}`,
    `=== TIER 1: FUND I SYLLABI (FOR BRIDGE QUESTIONS) ===\n\n${fundIDocs.join('\n\n---\n\n')}`,
    `=== TIER 2: IESE ENTREPRENEURSHIP ECOSYSTEM (USE WHEN CONTEXTUALLY RELEVANT) ===\n\n${tier2Docs.join('\n\n---\n\n')}`,
  ].join(separator);
}
```

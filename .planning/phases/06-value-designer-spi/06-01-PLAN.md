Key design decisions for these loaders:
- They do NOT join with megatrend docs — VD and SPI are facilitation engines that get no external knowledge base documents
- They do NOT cache (same pattern as getSystemContext which also reads fresh each call — Next.js module caching handles repeated calls)
- They use `path.join(process.cwd(), 'src', 'context', 'agents', ...)` — same path resolution pattern as the existing loader

The final `src/lib/context.ts` should look like this:

```typescript
import 'server-only';
import fs from 'fs';
import path from 'path';

export function getSystemContext(): string {
  const contextDir = path.join(process.cwd(), 'src', 'context');

  const systemPrompt = fs.readFileSync(
    path.join(contextDir, 'system-prompt.txt'),
    'utf-8'
  );

  const megatrendDocs = [1, 2, 3, 4].map((n) =>
    fs.readFileSync(path.join(contextDir, `megatrend-${n}.txt`), 'utf-8')
  );

  return [systemPrompt, ...megatrendDocs].join('\n\n---KNOWLEDGE BASE DOCUMENT---\n\n');
}

export function getValueDesignerContext(): string {
  const agentsDir = path.join(process.cwd(), 'src', 'context', 'agents');

export function getSPIContext(): string {
```

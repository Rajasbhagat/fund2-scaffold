```typescript
export const runtime = 'nodejs'

import { NextRequest, NextResponse } from 'next/server'
import path from 'path'
import fs from 'fs/promises'
import { randomUUID } from 'crypto'
import mammoth from 'mammoth'
import pdfParse from 'pdf-parse'
import JSZip from 'jszip'
import { prisma } from '@/lib/prisma'

const MAX_FILE_SIZE = 10 * 1024 * 1024 // 10 MB

const ALLOWED_MIME_TYPES: Record<string, string> = {
  'application/vnd.openxmlformats-officedocument.wordprocessingml.document': '.docx',
  'application/pdf': '.pdf',
  'application/vnd.openxmlformats-officedocument.presentationml.presentation': '.pptx',
}

async function extractText(buffer: Buffer, mimeType: string): Promise<string> {
  if (mimeType === 'application/vnd.openxmlformats-officedocument.wordprocessingml.document') {
    const result = await mammoth.extractRawText({ buffer })
    return result.value
  }

  if (mimeType === 'application/pdf') {
    const data = await pdfParse(buffer)
    return data.text
  }

  if (mimeType === 'application/vnd.openxmlformats-officedocument.presentationml.presentation') {
    return extractPptxText(buffer)
  }

  throw new Error(`Unsupported MIME type: ${mimeType}`)
}

async function extractPptxText(buffer: Buffer): Promise<string> {
  const zip = await JSZip.loadAsync(buffer)
  const slideFiles = Object.keys(zip.files)
    .filter((name) => /^ppt\/slides\/slide\d+\.xml$/.test(name))
    .sort()

  const texts: string[] = []
  for (const slideFile of slideFiles) {
    const xml = await zip.files[slideFile].async('text')
    const matches = xml.matchAll(/<a:t[^>]*>([^<]*)<\/a:t>/g)
    for (const match of matches) {
      const text = match[1].trim()
      if (text) texts.push(text)
    }
  }
  return texts.join(' ')
}

export async function POST(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id: projectId } = await params

  // Verify project exists
  const project = await prisma.project.findUnique({ where: { id: projectId } })
  if (!project) {
    return NextResponse.json({ error: 'Project not found' }, { status: 404 })
  }

  let formData: FormData
  try {
    formData = await request.formData()
  } catch {
    return NextResponse.json({ error: 'Invalid form data' }, { status: 400 })
  }

  const file = formData.get('file') as File | null
  if (!file) {
    return NextResponse.json({ error: 'No file provided' }, { status: 400 })
  }

  // Enforce 10 MB limit
  if (file.size > MAX_FILE_SIZE) {
    return NextResponse.json(
      { error: `File too large. Maximum size is 10 MB (received ${(file.size / 1024 / 1024).toFixed(1)} MB)` },
      { status: 400 }
    )
  }

  // Validate MIME type
  const ext = ALLOWED_MIME_TYPES[file.type]
  if (!ext) {
    return NextResponse.json(
      { error: `Unsupported file type: ${file.type}. Accepted: PDF, DOCX, PPTX` },
      { status: 400 }
    )
  }

  const buffer = Buffer.from(await file.arrayBuffer())

  // Extract text
  let extractedText: string
  try {
    extractedText = await extractText(buffer, file.type)
  } catch (err) {
    const message = err instanceof Error ? err.message : 'Extraction failed'
    return NextResponse.json({ error: `Text extraction failed: ${message}` }, { status: 422 })
  }

  // Save file to disk
  const uploadsDir = path.join(process.cwd(), 'uploads', projectId)
  await fs.mkdir(uploadsDir, { recursive: true })
  const filename = `${randomUUID()}${ext}`
  const filePath = path.join(uploadsDir, filename)
  await fs.writeFile(filePath, buffer)

  // Save record to DB
  const uploaded = await prisma.uploadedFile.create({
    data: {
      projectId,
      filename,
      originalName: file.name,
      mimeType: file.type,
      extractedText,
    },
    select: { id: true, originalName: true, mimeType: true, createdAt: true },
  })

  return NextResponse.json(uploaded, { status: 201 })
}

export async function GET(
  _request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id: projectId } = await params

  const project = await prisma.project.findUnique({ where: { id: projectId } })
  if (!project) {
    return NextResponse.json({ error: 'Project not found' }, { status: 404 })
  }

  const files = await prisma.uploadedFile.findMany({
    where: { projectId },
    select: { id: true, originalName: true, mimeType: true, createdAt: true },
    orderBy: { createdAt: 'asc' },
  })

  return NextResponse.json(files)
}
```

Key implementation notes:
- `export const runtime = 'nodejs'` MUST be the first line — pdf-parse and mammoth both require Node.js runtime (not Edge)
- `extractedText` is NOT included in the GET response (`select` explicitly excludes it) — it can be large and the UI only needs the filename
- The POST response uses status 201 (Created), not 200
- File size check happens BEFORE `arrayBuffer()` to avoid unnecessary memory allocation for oversized files
- `ALLOWED_MIME_TYPES` maps MIME type → extension. The extension is used for the stored filename so file type is visible on disk
- PPTX extraction: `ppt/slides/slide\d+\.xml` regex matches slide1.xml, slide2.xml etc. The `.sort()` ensures reading in order
- On extraction failure, return 422 (Unprocessable Entity) — the file type is valid but content could not be processed

```typescript
export const runtime = 'nodejs'

import { NextRequest, NextResponse } from 'next/server'
import path from 'path'
import fs from 'fs/promises'
import { prisma } from '@/lib/prisma'

export async function DELETE(
  _request: NextRequest,
  { params }: { params: Promise<{ id: string; fileId: string }> }
) {
  const { id: projectId, fileId } = await params

  // Find the record first to get the filename for disk deletion
  const uploadedFile = await prisma.uploadedFile.findUnique({
    where: { id: fileId },
  })

  if (!uploadedFile) {
    return NextResponse.json({ error: 'File not found' }, { status: 404 })
  }

  // Verify the file belongs to the specified project (prevents cross-project deletion)
  if (uploadedFile.projectId !== projectId) {
    return NextResponse.json({ error: 'File not found' }, { status: 404 })
  }

  // Delete from disk first (best-effort — don't fail if already missing)
  const filePath = path.join(process.cwd(), 'uploads', projectId, uploadedFile.filename)
  await fs.unlink(filePath).catch(() => {
    // File may have been manually deleted — ignore error, proceed with DB deletion
  })

  // Delete from DB
  await prisma.uploadedFile.delete({ where: { id: fileId } })

  return new NextResponse(null, { status: 204 })
}
```

Key implementation notes:
- Returns `204 No Content` (not 200 with body) — standard REST convention for successful DELETE
- `new NextResponse(null, { status: 204 })` is the correct pattern for a body-less response in Next.js App Router
- Cross-project ownership check: verify `uploadedFile.projectId === projectId` before deleting — returns 404 either way to avoid leaking existence information
- `fs.unlink(...).catch(() => {})` — silently ignores missing file on disk (idempotent cleanup)
- Both DB and disk deletion are attempted; DB deletion is the authoritative operation
- `export const runtime = 'nodejs'` required for `fs` module access
